    # 발차기 - 다리 전방 + 회전

    # 회전 공격 - 몸통 회전

### 3. Player 통합 (Day 17)

```gdscript
# Player.gd에서
func use_martial_art(slot: int) -> void:
    var martial = get_martial_in_slot(slot)
    if not martial:
        return
    
    # 애니메이션 시작
    animation_controller.play_animation(
        "martial_" + str(martial.id),
        martial.duration,
        false,  # 반복 아님
        Callable(self, "_on_martial_animation_complete")
    )
    
    # 에너지 소모
    current_energy -= martial.energy_cost

func _on_martial_animation_complete() -> void:
    # 데미지 계산 & 적에게 전달
    var targets = find_targets_in_range()
    for target in targets:
        target.take_damage(calculate_martial_damage(current_martial))
```
